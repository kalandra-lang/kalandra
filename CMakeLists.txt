cmake_minimum_required(VERSION 3.28)

file(READ "${CMAKE_CURRENT_LIST_DIR}/native/toolchain-proof/toolchain.json" KALANDRA_TOOLCHAIN_LOCK)
string(JSON KALANDRA_LLVM_VERSION GET "${KALANDRA_TOOLCHAIN_LOCK}" llvmVersion)
include("${CMAKE_CURRENT_LIST_DIR}/native/cmake/DevelopmentToolchain.cmake")
string(JSON KALANDRA_DEFAULT_LLVM_ROOT GET "${KALANDRA_TOOLCHAIN_LOCK}" defaultLlvmRoot)
if(DEFINED ENV{KALANDRA_LLVM_ROOT})
  set(KALANDRA_DEFAULT_LLVM_ROOT "$ENV{KALANDRA_LLVM_ROOT}")
endif()
set(KALANDRA_LLVM_ROOT "${KALANDRA_DEFAULT_LLVM_ROOT}" CACHE PATH "Pinned LLVM/MLIR installation")
set(_compiler_suffix "")
if(CMAKE_HOST_WIN32)
  set(_compiler_suffix ".exe")
  set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded CACHE STRING "Match LLVM's static Microsoft CRT")
endif()
set(CMAKE_AR "${KALANDRA_LLVM_ROOT}/bin/llvm-ar${_compiler_suffix}" CACHE FILEPATH "Pinned deterministic archiver")
set(CMAKE_RANLIB "${KALANDRA_LLVM_ROOT}/bin/llvm-ranlib${_compiler_suffix}" CACHE FILEPATH "Pinned deterministic archive indexer")
if(NOT DEFINED CMAKE_C_COMPILER)
  set(CMAKE_C_COMPILER "${KALANDRA_LLVM_ROOT}/bin/clang${_compiler_suffix}" CACHE FILEPATH "C compiler")
endif()
if(NOT DEFINED CMAKE_CXX_COMPILER)
  set(CMAKE_CXX_COMPILER "${KALANDRA_LLVM_ROOT}/bin/clang++${_compiler_suffix}" CACHE FILEPATH "C++ compiler")
endif()
if(CMAKE_HOST_APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "26.0" CACHE STRING "Minimum supported macOS")
  if(CMAKE_OSX_DEPLOYMENT_TARGET VERSION_LESS "26.0")
    message(FATAL_ERROR "Native distributions require macOS 26 or later")
  endif()
endif()
include(native/cmake/KalandraVersion.cmake)
project(KalandraNative VERSION ${KALANDRA_VERSION_CORE} LANGUAGES C CXX)
configure_file(native/include/kalandra/Version.h.in
  "${PROJECT_BINARY_DIR}/generated/kalandra/Version.h" @ONLY)
add_library(kalandra-version INTERFACE)
target_include_directories(kalandra-version INTERFACE "${PROJECT_BINARY_DIR}/generated")
set(_native_target_options)
if(APPLE)
  list(APPEND _native_target_options "-mmacosx-version-min=${CMAKE_OSX_DEPLOYMENT_TARGET}")
  add_compile_definitions(KALANDRA_NATIVE_MACOS_MINIMUM="${CMAKE_OSX_DEPLOYMENT_TARGET}")
endif()
add_compile_options("-ffile-prefix-map=${PROJECT_SOURCE_DIR}=/kalandra/src"
  "-ffile-prefix-map=${PROJECT_BINARY_DIR}=/kalandra/build"
  "-fdebug-compilation-dir=/kalandra/build")
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(APPLE AND CMAKE_SYSTEM_PROCESSOR STREQUAL "arm64")
  set(KALANDRA_NATIVE_PLATFORM "macos-arm64")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
  set(KALANDRA_NATIVE_PLATFORM "linux-x86_64")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
  set(KALANDRA_NATIVE_PLATFORM "linux-arm64")
elseif(WIN32 AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
  set(KALANDRA_NATIVE_PLATFORM "windows-x86_64")
elseif(WIN32 AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
  set(KALANDRA_NATIVE_PLATFORM "windows-arm64")
else()
  message(FATAL_ERROR "Native qualification targets macOS ARM64, Linux x86-64/ARM64 and Windows x86-64/ARM64")
endif()
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR
   NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL KALANDRA_LLVM_VERSION OR
   NOT CMAKE_C_COMPILER_VERSION VERSION_EQUAL KALANDRA_LLVM_VERSION)
  message(FATAL_ERROR "Use Clang ${KALANDRA_LLVM_VERSION} from KALANDRA_LLVM_ROOT")
endif()
file(WRITE "${PROJECT_BINARY_DIR}/CMakeFiles/kalandra-native-target.cpp" "int kalandra_native_target;\n")
execute_process(COMMAND "${CMAKE_CXX_COMPILER}" ${_native_target_options} -S -emit-llvm -x c++
  "${PROJECT_BINARY_DIR}/CMakeFiles/kalandra-native-target.cpp" -o -
  OUTPUT_VARIABLE _native_ir ERROR_VARIABLE _native_error RESULT_VARIABLE _native_status)
if(NOT _native_status EQUAL 0)
  message(FATAL_ERROR "Cannot query native target ABI: ${_native_error}")
endif()
string(REGEX MATCH "target datalayout = \"([^\"]+)\"" _layout "${_native_ir}")
set(KALANDRA_NATIVE_CPU_DATA_LAYOUT "${CMAKE_MATCH_1}")
string(REGEX MATCH "target triple = \"([^\"]+)\"" _triple "${_native_ir}")
set(KALANDRA_NATIVE_CPU_TARGET_TRIPLE "${CMAKE_MATCH_1}")
if(NOT KALANDRA_NATIVE_CPU_DATA_LAYOUT OR NOT KALANDRA_NATIVE_CPU_TARGET_TRIPLE)
  message(FATAL_ERROR "Pinned Clang did not report its native ABI")
endif()
add_compile_definitions(KALANDRA_NATIVE_CPU_DATA_LAYOUT="${KALANDRA_NATIVE_CPU_DATA_LAYOUT}"
  KALANDRA_NATIVE_CPU_TARGET_TRIPLE="${KALANDRA_NATIVE_CPU_TARGET_TRIPLE}")
if(WIN32)
  # Standard C environment reads are intentional; retain all ordinary warnings.
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
  # COFF objects otherwise carry wall-clock timestamps, including inside .lib.
  add_compile_options(-mno-incremental-linker-compatible -gno-codeview-command-line)
  # CodeView's S_OBJNAME does not honor source/debug prefix maps. Keep each
  # object's target-relative identity without embedding the physical build root.
  foreach(_language IN ITEMS C CXX)
    string(APPEND CMAKE_${_language}_COMPILE_OBJECT
      " -Xclang -object-file-name=/kalandra/build/<OBJECT>")
  endforeach()
  execute_process(COMMAND "${CMAKE_CXX_COMPILER}" --rtlib=compiler-rt -print-libgcc-file-name
    OUTPUT_VARIABLE KALANDRA_COMPILER_BUILTINS OUTPUT_STRIP_TRAILING_WHITESPACE
    RESULT_VARIABLE _builtins_status)
  if(NOT _builtins_status EQUAL 0 OR NOT EXISTS "${KALANDRA_COMPILER_BUILTINS}")
    message(FATAL_ERROR "Pinned Windows Clang lacks compiler-rt builtins: ${KALANDRA_COMPILER_BUILTINS}")
  endif()
  # CMake's MSVC-target link rule uses -nostdlib, so supply compiler helpers
  # explicitly (including the checker's exact 128-bit constant arithmetic).
  link_libraries("${KALANDRA_COMPILER_BUILTINS}")
  add_link_options("LINKER:/Brepro" "LINKER:/MANIFEST:EMBED"
    "LINKER:/MANIFESTINPUT:${PROJECT_SOURCE_DIR}/native/windows/application.manifest")
endif()
include(CTest)
add_subdirectory(native)
