# Standalone bring-up avoids installing the compiler's LLVM/MLIR dependencies
# on qualification hosts. Production compiler builds retain their pinned toolchain.
cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(KalandraVulkanBringup LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 23)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  include(CTest)
endif()
find_package(Vulkan REQUIRED COMPONENTS glslc)
find_package(Threads REQUIRED)
find_program(KALANDRA_SPIRV_VAL spirv-val REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(KALANDRA_VULKAN_SANITIZERS "Enable address/undefined sanitizers for bring-up" OFF)
option(KALANDRA_VULKAN_THREAD_SANITIZER "Enable thread sanitizer for bring-up" OFF)
if(KALANDRA_VULKAN_SANITIZERS AND KALANDRA_VULKAN_THREAD_SANITIZER)
  message(FATAL_ERROR "Address/undefined and thread sanitizers need separate build directories")
endif()
set(_vulkan_sanitizers)
if(KALANDRA_VULKAN_THREAD_SANITIZER)
  set(_vulkan_sanitizers -fsanitize=thread)
elseif(KALANDRA_VULKAN_SANITIZERS)
  set(_vulkan_sanitizers -fsanitize=address,undefined -fno-sanitize-recover=all)
endif()
set(_shaders "${CMAKE_CURRENT_SOURCE_DIR}/third_party/ggml")
file(GLOB _shader_dependencies CONFIGURE_DEPENDS "${_shaders}/*.glsl")
foreach(_shader IN ITEMS add mul_mm)
  set(_output "${CMAKE_CURRENT_BINARY_DIR}/${_shader}.spv")
  add_custom_command(OUTPUT "${_output}"
    COMMAND "${Vulkan_GLSLC_EXECUTABLE}" --target-env=vulkan1.1 -O -I "${_shaders}"
      -DDATA_A_F32=1 -DB_TYPE=float -DB_TYPE_SCALAR=float -DD_TYPE=float
      -DLOAD_VEC_A=1 -DLOAD_VEC_B=1 -DFLOAT_TYPE=float
      -DFLOAT_TYPEV2=vec2 -DFLOAT_TYPEV4=vec4 -DACC_TYPE=float -DACC_TYPEV2=vec2
      "${_shaders}/${_shader}.comp" -o "${_output}"
    COMMAND "${KALANDRA_SPIRV_VAL}" --target-env vulkan1.1 "${_output}"
    DEPENDS "${_shaders}/${_shader}.comp" ${_shader_dependencies}
    VERBATIM)
  list(APPEND _outputs "${_output}")
endforeach()
set(_transform_sources)
foreach(_type IN ITEMS int8 uint8 int16 uint16 int32 uint32 int64 uint64 float32 float64)
  list(APPEND _transform_sources "${CMAKE_CURRENT_BINARY_DIR}/transforms/transpose-${_type}.comp")
endforeach()
foreach(_kind IN ITEMS softmaxRows rmsNormRows)
  list(APPEND _transform_sources "${CMAKE_CURRENT_BINARY_DIR}/transforms/${_kind}-float32.comp")
endforeach()
add_custom_command(OUTPUT ${_transform_sources}
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/GenerateTransforms.py"
    --upstream "${_shaders}" --metal "${CMAKE_CURRENT_SOURCE_DIR}/../metal-source/TensorTransformShader.h"
    --output "${CMAKE_CURRENT_BINARY_DIR}/transforms"
  DEPENDS GenerateTransforms.py "${_shaders}/copy_transpose.comp"
    "${CMAKE_CURRENT_SOURCE_DIR}/../metal-source/TensorTransformShader.h"
  VERBATIM)
foreach(_source IN LISTS _transform_sources)
  get_filename_component(_name "${_source}" NAME_WE)
  string(REPLACE "-" ";" _parts "${_name}")
  list(GET _parts 0 _kind)
  list(GET _parts 1 _type)
  set(_output "${CMAKE_CURRENT_BINARY_DIR}/transforms/${_name}.spv")
  set(_transform_environment vulkan1.1)
  if(NOT _kind STREQUAL "transpose")
    set(_transform_environment vulkan1.2)
  endif()
  add_custom_command(OUTPUT "${_output}"
    COMMAND "${Vulkan_GLSLC_EXECUTABLE}" --target-env=vulkan1.1 -O "${_source}" -o "${_output}.unnamed"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/RenameTransformEntry.py"
      "${_output}.unnamed" "${_output}" "${_kind}" "${_type}"
    COMMAND "${KALANDRA_SPIRV_VAL}" --target-env "${_transform_environment}" "${_output}"
    DEPENDS "${_source}" RenameTransformEntry.py
    BYPRODUCTS "${_output}.unnamed" VERBATIM)
  list(APPEND _outputs "${_output}")
endforeach()
add_custom_target(kalandra-vulkan-shaders DEPENDS ${_outputs})
set(_embedded_shaders "${CMAKE_CURRENT_BINARY_DIR}/BuiltinShaders.cpp")
add_custom_command(OUTPUT "${_embedded_shaders}"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/EmbedShaders.py"
    --root "${CMAKE_CURRENT_BINARY_DIR}" --output "${_embedded_shaders}" ${_outputs}
  DEPENDS ${_outputs} EmbedShaders.py VERBATIM)
# The object library has no Vulkan/OS dependency. Include its objects in the raw
# runtime archive as well, so consumers do not need an extra link-order contract.
add_library(kalandra-vulkan-resources OBJECT "${_embedded_shaders}")
target_include_directories(kalandra-vulkan-resources PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(kalandra-vulkan STATIC Vulkan.cpp SpirvInterface.cpp)
target_link_libraries(kalandra-vulkan PRIVATE kalandra-vulkan-resources)
target_link_libraries(kalandra-vulkan PUBLIC Vulkan::Vulkan Threads::Threads)
target_include_directories(kalandra-vulkan PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(kalandra-vulkan PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
add_dependencies(kalandra-vulkan kalandra-vulkan-shaders)
add_library(kalandra-vulkan-program STATIC Program.cpp TensorBridge.cpp)
target_include_directories(kalandra-vulkan-program PUBLIC ../include)
target_link_libraries(kalandra-vulkan-program PUBLIC kalandra-vulkan)
target_compile_options(kalandra-vulkan-program PRIVATE -Wall -Wextra -Werror)
if(_vulkan_sanitizers)
  target_compile_options(kalandra-vulkan PUBLIC ${_vulkan_sanitizers} -fno-omit-frame-pointer)
  target_link_options(kalandra-vulkan PUBLIC ${_vulkan_sanitizers})
endif()

# Managed dispatch uses the existing task core. The standalone project can
# compile that core from the repository without LLVM/MLIR dependencies.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../runtime/Tasks.cpp")

    set(_vulkan_task_core kalandra-native-runtime)

      set(_vulkan_task_core kalandra-native-runtime-sanitized)
    elseif(KALANDRA_VULKAN_THREAD_SANITIZER)
      set(_vulkan_task_core kalandra-native-runtime-thread-sanitized)
    endif()
  else()
    add_library(kalandra-vulkan-task-core STATIC ../runtime/Runtime.cpp ../runtime/Tasks.cpp)
    target_link_libraries(kalandra-vulkan-task-core PUBLIC Threads::Threads)
    target_compile_options(kalandra-vulkan-task-core PRIVATE -Wall -Wextra -Werror)
    if(_vulkan_sanitizers)
      target_compile_options(kalandra-vulkan-task-core PRIVATE ${_vulkan_sanitizers} -fno-omit-frame-pointer)
      target_link_options(kalandra-vulkan-task-core PUBLIC ${_vulkan_sanitizers})
    endif()
    set(_vulkan_task_core kalandra-vulkan-task-core)
  endif()
  target_link_libraries(kalandra-vulkan-program PUBLIC ${_vulkan_task_core})
  add_library(kalandra-vulkan-async STATIC Async.cpp)
  target_link_libraries(kalandra-vulkan-async PUBLIC kalandra-vulkan-program ${_vulkan_task_core})
  target_compile_options(kalandra-vulkan-async PRIVATE -Wall -Wextra -Werror)
  add_library(kalandra-vulkan-language STATIC Language.cpp)
  target_link_libraries(kalandra-vulkan-language PUBLIC kalandra-vulkan-async)
  target_compile_options(kalandra-vulkan-language PRIVATE -Wall -Wextra -Werror)
  if(TARGET kalandra-native-runtime-sanitized AND NOT _vulkan_sanitizers AND NOT WIN32)
