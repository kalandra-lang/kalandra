# Standalone bring-up avoids installing the compiler's LLVM/MLIR dependencies
# on qualification hosts. Production compiler builds retain their pinned toolchain.
cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(KalandraVulkanBringup LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 23)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  include(CTest)
endif()
find_package(Vulkan REQUIRED COMPONENTS glslc)
find_package(Threads REQUIRED)
find_program(KALANDRA_SPIRV_VAL spirv-val REQUIRED)
option(KALANDRA_VULKAN_SANITIZERS "Enable address/undefined sanitizers for bring-up" OFF)
option(KALANDRA_VULKAN_THREAD_SANITIZER "Enable thread sanitizer for bring-up" OFF)
if(KALANDRA_VULKAN_SANITIZERS AND KALANDRA_VULKAN_THREAD_SANITIZER)
  message(FATAL_ERROR "Address/undefined and thread sanitizers need separate build directories")
endif()
set(_vulkan_sanitizers)
if(KALANDRA_VULKAN_THREAD_SANITIZER)
  set(_vulkan_sanitizers -fsanitize=thread)
elseif(KALANDRA_VULKAN_SANITIZERS)
  set(_vulkan_sanitizers -fsanitize=address,undefined -fno-sanitize-recover=all)
endif()
set(_shaders "${CMAKE_CURRENT_SOURCE_DIR}/third_party/ggml")
file(GLOB _shader_dependencies CONFIGURE_DEPENDS "${_shaders}/*.glsl")
foreach(_shader IN ITEMS add mul_mm)
  set(_output "${CMAKE_CURRENT_BINARY_DIR}/${_shader}.spv")
  add_custom_command(OUTPUT "${_output}"
    COMMAND "${Vulkan_GLSLC_EXECUTABLE}" --target-env=vulkan1.1 -O -I "${_shaders}"
      -DDATA_A_F32=1 -DB_TYPE=float -DB_TYPE_SCALAR=float -DD_TYPE=float
      -DLOAD_VEC_A=1 -DLOAD_VEC_B=1 -DFLOAT_TYPE=float
      -DFLOAT_TYPEV2=vec2 -DFLOAT_TYPEV4=vec4 -DACC_TYPE=float -DACC_TYPEV2=vec2
      "${_shaders}/${_shader}.comp" -o "${_output}"
    COMMAND "${KALANDRA_SPIRV_VAL}" --target-env vulkan1.1 "${_output}"
    DEPENDS "${_shaders}/${_shader}.comp" ${_shader_dependencies}
    VERBATIM)
  list(APPEND _outputs "${_output}")
endforeach()
add_custom_target(kalandra-vulkan-shaders DEPENDS ${_outputs})
add_library(kalandra-vulkan STATIC Vulkan.cpp SpirvInterface.cpp)
target_link_libraries(kalandra-vulkan PUBLIC Vulkan::Vulkan Threads::Threads)
target_include_directories(kalandra-vulkan PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(kalandra-vulkan PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
# Temporary bring-up artifact paths. Release packaging/relocation is not implemented.
target_compile_definitions(kalandra-vulkan PRIVATE
  KALANDRA_VULKAN_SHADER_DIR="${CMAKE_CURRENT_BINARY_DIR}")
add_dependencies(kalandra-vulkan kalandra-vulkan-shaders)
if(_vulkan_sanitizers)
  target_compile_options(kalandra-vulkan PUBLIC ${_vulkan_sanitizers} -fno-omit-frame-pointer)
  target_link_options(kalandra-vulkan PUBLIC ${_vulkan_sanitizers})
endif()
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  find_program(KALANDRA_SPIRV_DIS spirv-dis REQUIRED)
  add_test(NAME native-vulkan-upstream-audit
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/audit.py"
      "${CMAKE_CURRENT_BINARY_DIR}" --spirv-dis "${KALANDRA_SPIRV_DIS}")
  add_test(NAME native-vulkan-development-toolchain
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_development_toolchain.py")
  add_executable(kalandra-vulkan-typed-kernel-tests tests/TypedKernelTests.cpp)
  add_executable(kalandra-vulkan-primitive-kernel-tests tests/PrimitiveKernelTests.cpp)
  target_include_directories(kalandra-vulkan-primitive-kernel-tests PRIVATE ../include)
  target_compile_options(kalandra-vulkan-primitive-kernel-tests PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
  target_link_libraries(kalandra-vulkan-primitive-kernel-tests PRIVATE kalandra-vulkan)
  target_compile_options(kalandra-vulkan-typed-kernel-tests PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(kalandra-vulkan-typed-kernel-tests PRIVATE kalandra-vulkan)
  add_executable(kalandra-vulkan-compiler-kernel-tests tests/CompilerKernelTests.cpp)
  target_compile_options(kalandra-vulkan-compiler-kernel-tests PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
  target_link_libraries(kalandra-vulkan-compiler-kernel-tests PRIVATE kalandra-vulkan)
  add_executable(kalandra-vulkan-interface-tests tests/SpirvInterfaceTests.cpp SpirvInterface.cpp)
  target_include_directories(kalandra-vulkan-interface-tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_options(kalandra-vulkan-interface-tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME native-vulkan-interface COMMAND kalandra-vulkan-interface-tests)
  add_executable(kalandra-vulkan-loader-control tests/LoaderControl.cpp)
  target_link_libraries(kalandra-vulkan-loader-control PRIVATE Vulkan::Vulkan)
  if(_vulkan_sanitizers)
    target_compile_options(kalandra-vulkan-loader-control PRIVATE ${_vulkan_sanitizers} -fno-omit-frame-pointer)
    target_link_options(kalandra-vulkan-loader-control PRIVATE ${_vulkan_sanitizers})
  endif()
  add_executable(kalandra-vulkan-tests tests/RuntimeTests.cpp)
  target_compile_options(kalandra-vulkan-tests PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(kalandra-vulkan-tests PRIVATE kalandra-vulkan)
  add_test(NAME native-vulkan-runtime COMMAND kalandra-vulkan-tests)
  set_tests_properties(native-vulkan-runtime PROPERTIES TIMEOUT 120 RESOURCE_LOCK vulkan-gpu)
endif()

# Managed dispatch uses the existing task core. The standalone project can
# compile that core from the repository without LLVM/MLIR dependencies.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../runtime/Tasks.cpp")
  if(TARGET kalandra-native-runtime)
    set(_vulkan_task_core kalandra-native-runtime)
    if(KALANDRA_VULKAN_SANITIZERS)
      set(_vulkan_task_core kalandra-native-runtime-sanitized)
    elseif(KALANDRA_VULKAN_THREAD_SANITIZER)
      set(_vulkan_task_core kalandra-native-runtime-thread-sanitized)
    endif()
  else()
    add_library(kalandra-vulkan-task-core STATIC ../runtime/Runtime.cpp ../runtime/Tasks.cpp)
    target_link_libraries(kalandra-vulkan-task-core PUBLIC Threads::Threads)
    target_compile_options(kalandra-vulkan-task-core PRIVATE -Wall -Wextra -Werror)
    if(_vulkan_sanitizers)
      target_compile_options(kalandra-vulkan-task-core PRIVATE ${_vulkan_sanitizers} -fno-omit-frame-pointer)
      target_link_options(kalandra-vulkan-task-core PUBLIC ${_vulkan_sanitizers})
    endif()
    set(_vulkan_task_core kalandra-vulkan-task-core)
  endif()
  add_library(kalandra-vulkan-async STATIC Async.cpp)
  target_link_libraries(kalandra-vulkan-async PUBLIC kalandra-vulkan ${_vulkan_task_core})
  target_compile_options(kalandra-vulkan-async PRIVATE -Wall -Wextra -Werror)
  if(BUILD_TESTING)
    add_executable(kalandra-vulkan-async-tests tests/AsyncTests.cpp)
    target_link_libraries(kalandra-vulkan-async-tests PRIVATE kalandra-vulkan-async)
    target_compile_options(kalandra-vulkan-async-tests PRIVATE -Wall -Wextra -Werror)
    set(KALANDRA_VULKAN_TEST_KERNEL "" CACHE FILEPATH "Source-generated combine kernel for managed Vulkan tests")
    if(KALANDRA_VULKAN_TEST_KERNEL)
      add_test(NAME native-vulkan-async COMMAND kalandra-vulkan-async-tests "${KALANDRA_VULKAN_TEST_KERNEL}")
      set_tests_properties(native-vulkan-async PROPERTIES TIMEOUT 120 RESOURCE_LOCK vulkan-gpu)
    endif()
  endif()
endif()
